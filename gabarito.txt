 GABARITO DAS MISSÕES - SQL QUEST
=========================================

--- NÍVEL 1: Fundamentos de Extração ---

1.1 Conhecendo a Base
SELECT * FROM alunos;

1.2 Focando no Essencial
SELECT nome, turno FROM alunos;

1.3 O Primeiro Filtro
SELECT * FROM alunos WHERE turno = 'Noite';


--- NÍVEL 2: Refinando Buscas ---

2.1 Filtro Duplo (AND)
SELECT * FROM alunos WHERE turno = 'Noite' AND data_nascimento < '2000-01-01';

2.2 Lista de Chamada
SELECT nome FROM alunos ORDER BY nome ASC;

2.3 O Pódio (LIMIT)
SELECT aluno_id, nota FROM matriculas ORDER BY nota DESC LIMIT 3;


--- NÍVEL 3: A Matemática do Banco ---

3.1 Contagem de Alunos
SELECT COUNT(*) FROM alunos;

3.2 Carga Horária Total
SELECT SUM(carga_horaria) FROM disciplinas;

3.3 Os Extremos (MAX / MIN)
SELECT MAX(nota), MIN(nota) FROM matriculas;

3.4 Média de Notas (AVG)
SELECT AVG(nota) FROM matriculas;


--- NÍVEL 4: Organizando Resultados ---

4.1 Alunos por Turno
SELECT turno, COUNT(*) FROM alunos GROUP BY turno;

4.2 Desempenho por Matéria
SELECT disciplina_id, AVG(nota) FROM matriculas GROUP BY disciplina_id;

4.3 Investigação (HAVING)
SELECT disciplina_id, AVG(nota) FROM matriculas GROUP BY disciplina_id HAVING AVG(nota) < 6.0;


--- NÍVEL 5: Juntando as Peças (JOINS) ---

5.1 O Nome da Disciplina
SELECT matriculas.nota, disciplinas.nome_disciplina 
FROM matriculas 
INNER JOIN disciplinas ON matriculas.disciplina_id = disciplinas.id;

5.2 Quem cursou o quê?
SELECT alunos.nome, matriculas.nota 
FROM alunos 
INNER JOIN matriculas ON alunos.id = matriculas.aluno_id;

5.3 O Boletim Completo
SELECT alunos.nome, disciplinas.nome_disciplina, matriculas.nota 
FROM alunos 
INNER JOIN matriculas ON alunos.id = matriculas.aluno_id 
INNER JOIN disciplinas ON matriculas.disciplina_id = disciplinas.id;
